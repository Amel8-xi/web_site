function initMap() {
  if (leafletMap) return;
  const mapEl = document.getElementById('map-container');
  if (!mapEl) return;

  leafletMap = L.map('map-container').setView([29.8, -1.9], 7);

  L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
    maxZoom: 18,
    attribution: '© OpenStreetMap'
  }).addTo(leafletMap);

  updateMapMarkers();
}

function updateMapMarkers() {
  if (!leafletMap) return;
  mapMarkers.forEach(m => leafletMap.removeLayer(m));
  mapMarkers = [];

  localCampaigns.filter(c => c.status === "نشطة" || c.status === "معتمدة").forEach(camp => {
    const lat = camp.lat || (MUNICIPALITY_COORDS[camp.city] ? MUNICIPALITY_COORDS[camp.city][0] : 30.1333);
    const lng = camp.lng || (MUNICIPALITY_COORDS[camp.city] ? MUNICIPALITY_COORDS[camp.city][1] : -2.1667);
    
    const marker = L.circleMarker([lat, lng], {
      radius: 10,
      fillColor: "#0f8f74",
      color: "#ffffff",
      weight: 2,
      opacity: 1,
      fillOpacity: 0.95
    }).addTo(leafletMap);

    const popupContent = `
      <div style="font-family:'Tajawal',sans-serif; text-align:right;">
        <h4 style="margin:0 0 5px; color:#0d382d; font-size:14px; font-weight:800;">${camp.title}</h4>
        <p style="margin:0 0 4px; font-size:12px; color:#15803d; font-weight:700;">📍 بلدية ${camp.city}</p>
        <p style="margin:0 0 4px; font-size:11px; color:#0369a1; font-weight:700;">📅 التاريخ: ${camp.campDate || 'غير محدد'}</p>
        <p style="margin:0 0 4px; font-size:11px; color:#1e293b; font-weight:bold;">👤 المنشئ: ${camp.creatorName || 'غير معروف'} (العمر: ${camp.creatorAge || '-'})</p>
        <p style="margin:0 0 6px; font-size:11px; color:#64748b;">المجال: ${camp.skill} (${camp.urgency})</p>
      </div>
    `;
    marker.bindPopup(popupContent);
    mapMarkers.push(marker);
  });
}

function initPickerMap() {
  const city = document.getElementById('camp-city').value;
  const defaultCoords = MUNICIPALITY_COORDS[city] || [30.1333, -2.1667];

  if (!pickerMap) {
    pickerMap = L.map('picker-map-container').setView(defaultCoords, 11);
    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
      maxZoom: 18,
      attribution: '© OpenStreetMap'
    }).addTo(pickerMap);

    pickerMarker = L.marker(defaultCoords, { draggable: true }).addTo(pickerMap);

    pickerMarker.on('dragend', function(e) {
      const pos = e.target.getLatLng();
      setCampaignPickerCoords(pos.lat, pos.lng);
    });

    pickerMap.on('click', function(e) {
      pickerMarker.setLatLng(e.latlng);
      setCampaignPickerCoords(e.latlng.lat, e.latlng.lng);
    });
  } else {
    pickerMap.invalidateSize();
    pickerMap.setView(defaultCoords, 11);
    pickerMarker.setLatLng(defaultCoords);
    setCampaignPickerCoords(defaultCoords[0], defaultCoords[1]);
  }
}

function setCampaignPickerCoords(lat, lng) {
  document.getElementById('camp-lat').value = lat.toFixed(6);
  document.getElementById('camp-lng').value = lng.toFixed(6);
  const city = document.getElementById('camp-city').value;
  document.getElementById('picker-coords-text').textContent = 
    `الموقع المحدد: بلدية ${city} (${lat.toFixed(4)}, ${lng.toFixed(4)})`;
}

window.onModalMunicipalityChange = function() {
  const city = document.getElementById('camp-city').value;
  const coords = MUNICIPALITY_COORDS[city] || [30.1333, -2.1667];
  if (pickerMap && pickerMarker) {
    pickerMap.setView(coords, 11);
    pickerMarker.setLatLng(coords);
    setCampaignPickerCoords(coords[0], coords[1]);
  }
};

window.locateUserGPSForCampaign = function() {
  if (navigator.geolocation) {
    navigator.geolocation.getCurrentPosition((pos) => {
      const lat = pos.coords.latitude;
      const lng = pos.coords.longitude;
      if (pickerMap && pickerMarker) {
        pickerMap.setView([lat, lng], 14);
        pickerMarker.setLatLng([lat, lng]);
        setCampaignPickerCoords(lat, lng);
        alert("تم التقاط موقعك الجغرافي وتثبيته على الخريطة بنجاح!");
      }
    }, () => {
      alert("تعذر الوصول لموقع GPS.");
    });
  }
};

window.focusOnMap = function(lat, lng) {
  switchMainTab('opps-section');
  if (leafletMap) {
    leafletMap.setView([lat, lng], 12, { animate: true });
    mapMarkers.forEach(m => {
      const pos = m.getLatLng();
      if (Math.abs(pos.lat - lat) < 0.001 && Math.abs(pos.lng - lng) < 0.001) {
        m.openPopup();
      }
    });
  }
};

window.switchMainTab = function(sectionId) {
  document.querySelectorAll('.app-section').forEach(sec => sec.classList.remove('active'));
  document.querySelectorAll('.nav-tab-btn').forEach(btn => btn.classList.remove('active'));

  const targetSec = document.getElementById(sectionId);
  if (targetSec) targetSec.classList.add('active');

  if (sectionId === 'home-section') {
    document.getElementById('tab-btn-home').classList.add('active');
  } else if (sectionId === 'opps-section') {
    document.getElementById('tab-btn-opps').classList.add('active');
    setTimeout(() => {
      initMap();
      if (leafletMap) {
        leafletMap.invalidateSize();
        updateMapMarkers();
      }
    }, 200);
  } else if (sectionId === 'attendance-section') {
    document.getElementById('tab-btn-attendance').classList.add('active');
  }
  window.scrollTo({ top: 0, behavior: 'smooth' });
};

function syncData() {
  if (useFirebase && db) {
    db.ref("volunteers").on("value", (snap) => {
      const val = snap.val();
      if (val) {
        localVolunteers = [];
        Object.keys(val).forEach(k => localVolunteers.push({ id: k, ...val[k] }));
        saveLocalData();
        updateAllUI();
      }
    });

    db.ref("campaigns").on("value", (snap) => {
      const val = snap.val();
      if (val) {
        localCampaigns = [];
        Object.keys(val).forEach(k => localCampaigns.push({ id: k, ...val[k] }));
        saveLocalData();
        updateAllUI();
      }
    });

    db.ref("attendance").on("value", (snap) => {
      const val = snap.val();
      if (val) {
        localAttendance = [];
        Object.keys(val).forEach(k => localAttendance.push({ id: k, ...val[k] }));
        saveLocalData();
        updateAllUI();
      }
    });

    db.ref("joins").on("value", (snap) => {
      const val = snap.val();
      if (val) {
        localJoins = [];
        Object.keys(val).forEach(k => localJoins.push({ id: k, ...val[k] }));
        saveLocalData();
        updateAllUI();
      }
    });
  }
}

function updateAllUI() {
  const volCount = document.getElementById("stat-volunteers");
  const campCount = document.getElementById("stat-campaigns");
  const hoursCount = document.getElementById("stat-hours");
  const badgesCount = document.getElementById("stat-badges");

  if (volCount) volCount.textContent = localVolunteers.length;
  if (campCount) campCount.textContent = localCampaigns.filter(c => c.status === "مكتملة").length;
  
  const totalHours = localAttendance.reduce((acc, curr) => acc + (Number(curr.hours) || 0), 0);
  if (hoursCount) hoursCount.textContent = totalHours;
  if (badgesCount) badgesCount.textContent = Math.floor(totalHours / 10);

  renderOpportunities();
  renderFeaturedCampaigns();
  updateVolunteerSelect();
  updateCampaignSelectUI();
  updateQuickAttendanceSelect();
  updateMatchingCampaignSelect();
  updateMapMarkers();
  if (document.getElementById("manager-dashboard-modal").style.display === "flex") {
    renderManagerCampaignsList();
    renderManagerVolunteersList();
  }
}

window.managerDeleteVolunteer = function(volId) {
  if (confirm("هل أنت متأكد من حذف هذا المتطوع نهائياً؟")) {
    localVolunteers = localVolunteers.filter(v => v.id !== volId);
    saveLocalData();
    if (useFirebase && db) {
      db.ref("volunteers/" + volId).remove().catch((e) => console.error(e));
    }
    updateAllUI();
    alert("تم حذف المتطوع بنجاح وإزالته من النظام.");
  }
};

window.managerApproveCampaign = function(id) {
  const camp = localCampaigns.find(c => c.id === id);
  if (camp) camp.status = "نشطة";
  saveLocalData();
  if (useFirebase && db) {
    db.ref("campaigns/" + id).update({ status: "نشطة" }).catch((e) => console.error(e));
  }
  updateAllUI();
  alert("تمت الموافقة على الحملة وأصبحت نشطة!");
};

window.clearAllCampaigns = function() {

window.applyToCampaign = function(campaignId, title) {
  const currentUser = localStorage.getItem("ba_logged_user");
  if (!currentUser) { alert("يرجى تسجيل الدخول أولاً!"); return; }

  const existingJoinIndex = localJoins.findIndex(j => j.campaignId === campaignId && j.user === currentUser);

  if (existingJoinIndex !== -1) {
    const joinedItem = localJoins[existingJoinIndex];
    localJoins.splice(existingJoinIndex, 1);
    saveLocalData();
    if (useFirebase && db && joinedItem.id) {
      db.ref("joins/" + joinedItem.id).remove();
    }
    updateAllUI();
    alert(`تم التراجع وإلغاء انضمامك إلى حملة "${title}" بنجاح.`);
  } else {
    const joinId = "join_" + Date.now();
    const joinData = { id: joinId, campaignId, campaignTitle: title, user: currentUser, timestamp: Date.now() };
    localJoins.push(joinData);
    saveLocalData();
    if (useFirebase && db) {
      db.ref("joins").child(joinId).set(joinData);
    }
    updateAllUI();
    alert(`تم بنجاح الانضمام إلى حملة "${title}"!`);
  }
};

window.runSmartMatching = function() {
  const campId = document.getElementById("matching-campaign-select").value;
  const resultsArea = document.getElementById("matching-results-area");
  if (!campId) return;

  const camp = localCampaigns.find(c => c.id === campId);
  if (!camp) return;

  const matched = localVolunteers.filter(v => v.city === camp.city || v.skill === camp.skill);
  if (matched.length === 0) {
    resultsArea.innerHTML = `<div style="text-align: center; color: #b91c1c; padding: 1.5rem; background:#fee2e2; border-radius: 10px;">لا يوجد متطوعون مطابقون حالياً.</div>`;
    return;
  }

  let html = `<h4 style="color: #0d382d; margin-bottom: 0.8rem;">المتطوعون المطابقون:</h4><div style="display: flex; flex-direction: column; gap: 0.8rem;">`;
  matched.forEach(v => {
    html += `
      <div style="background: #f8fafc; border: 1px solid #e2e8f0; padding: 1rem; border-radius: 8px; display: flex; justify-content: space-between; align-items: center;">
        <div><h5>${v.name}</h5><p style="font-size:0.8rem; color:#64748b;">بلدية: ${v.city} | التخصص: ${v.skill}</p></div>
        <button class="btn-apply" onclick="alert('تم تعيين المتطوع ${v.name} بنجاح')">تعيين بالفريق</button>
      </div>
    `;
  });
  html += `</div>`;
  resultsArea.innerHTML = html;
};

window.generateSelectedVolunteerQR = function() {
  const sel = document.getElementById("volunteer-select-qr");
  const canvas = document.getElementById("qrcode-canvas");
  const previewArea = document.getElementById("qr-preview-area");
  if (!sel || !canvas) return;

  const volId = sel.value;
  canvas.innerHTML = "";
  if (!volId) { if (previewArea) previewArea.style.display = "none"; return; }

  const vol = localVolunteers.find(v => v.id === volId);
  if (vol) {
    if (previewArea) previewArea.style.display = "flex";
    document.getElementById("qr-volunteer-name").textContent = vol.name;
    document.getElementById("qr-volunteer-municipality").textContent = `بلدية ${vol.city}`;
    document.getElementById("vol-info-phone").textContent = vol.phone || "غير متوفر";
    document.getElementById("vol-info-skill").textContent = vol.skill || "غير محدد";
    document.getElementById("vol-info-prac").textContent = vol.practicalSkills ? vol.practicalSkills.join(', ') : "لا توجد";
    document.getElementById("vol-info-hobbies").textContent = vol.hobbies ? vol.hobbies.join(', ') : "لا توجد";

    try {
      new QRCode(canvas, {
        text: `VOLUNTEER:${vol.id}:${vol.name}`,
        width: 160, height: 160, colorDark: "#000000", colorLight: "#ffffff", correctLevel: QRCode.CorrectLevel.M
      });
    } catch(e) {}
  }
};

window.startCameraScanner = function() {
  const camp = document.getElementById("att-campaign-select").value;
  if (!camp) { alert("اختر المبادرة أولاً!"); return; }

  const qrBox = document.getElementById("qr-reader");
  qrBox.innerHTML = "";

  html5QrCodeScanner = new Html5Qrcode("qr-reader");
  html5QrCodeScanner.start({ facingMode: "environment" }, { fps: 15, qrbox: 220 }, (text) => {
    handleQrDecoded(text);
  }).then(() => {
    document.getElementById("start-scanner-btn").style.display = "none";
    document.getElementById("stop-scanner-btn").style.display = "inline-flex";
  }).catch(() => alert("تعذر فتح الكاميرا."));
};

window.stopCameraScanner = function() {
  if (html5QrCodeScanner) {
    html5QrCodeScanner.stop().then(() => {
      document.getElementById("start-scanner-btn").style.display = "inline-flex";
      document.getElementById("stop-scanner-btn").style.display = "none";
      document.getElementById("qr-reader").innerHTML = "";
    }).catch(() => {});
  }
};

window.scanQRFromFile = function(input) {
  const camp = document.getElementById("att-campaign-select").value;
  if (!camp) { alert("اختر المبادرة أولاً!"); input.value = ""; return; }
  if (input.files && input.files.length > 0) {
    const html5QrCode = new Html5Qrcode("qr-reader");
    html5QrCode.scanFile(input.files[0], true)
      .then(text => { handleQrDecoded(text); input.value = ""; })
      .catch(() => { alert("تعذر قراءة الرمز."); input.value = ""; });
  }
};

function handleQrDecoded(text) {
  let volName = "متطوع";
  if (text.startsWith("VOLUNTEER:")) {
    volName = text.split(":")[2] || "متطوع";
  }
  recordAttendanceSuccess(volName);
  stopCameraScanner();
}
